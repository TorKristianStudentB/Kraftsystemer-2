==============================================================================
 KOMMANDOER FOR Å UNDERSØKE Grid-OBJEKTET
==============================================================================

Objektstruktur

  grid
   ├── .Name                 -> str,  navnet på nettet ("Grid")
   ├── .Base                 -> BaseValues-objekt
   │      ├── .Sbase         -> float, S_base [MVA]
   │      └── .Vbase         -> float, V_base [kV]
   │
   ├── .bus                  -> liste med bus-objekter
   │      └── [busindeks]
   │           ├── .busNumber     -> int,   busNumber (OBS: liten n = tallet)
   │           ├── .Name     -> str,   trafostasjon-navn (OBS: stor N = navnet)
   │           ├── .Volt     -> float, spenning [p.u.]
   │           ├── .Angle    -> float, vinkel [rad]
   │           ├── .P_gen    -> float
   │           ├── .Q_gen    -> float
   │           ├── .P_load   -> float
   │           ├── .Q_load   -> float
   │           ├── .bidz     -> str,   budområde (f.eks. "NO1")
   │           └── .Vbase    -> float, spenningsnivå [kV]
   │
   ├── .line                 -> liste med line-objekter
   │      └── [lineindeks]
   │           ├── .lineNumber   -> int
   │           ├── .Name         -> str, "bussfra – busstil"
   │           ├── .R            -> float
   │           ├── .X            -> float
   │           ├── .B            -> float
   │           ├── .Frombus      -> int
   │           ├── .Tobus        -> int
   │           ├── .lenght       -> float, lengde [m]
   │           └── .admittans()  -> complex, 1/(R+jX)
   │
   ├── .trafo                -> liste med trafo-objekter
   │      └── [trafoindeks]
   │           ├── .Name         -> str
   │           ├── .Type         -> str  ("tap")
   │           ├── .Ratio        -> float ("tror ikke denne stemmer")
   │           ├── .Frombus      -> int
   │           ├── .Tobus        -> int
   │           ├── .R            -> float
   │           ├── .X            -> float
   │           └── .admittans()  -> complex, 1/(R+jX)
   │
   ├── .gen                  -> liste med gen-objekter
   │      └── [genindeks]
   │           ├── .name         -> str,   generatornavn
   │           ├── .bus          -> float, buss-id generatoren står på
   │           ├── .P_max        -> int,   Pmax
   │           ├── .Q_max        -> float, Qmax (test)
   │           └── .Q_min        -> float, Qmin (test)
   │
   ├── .thevenin(bus1, bus2)      -> Thevenin-impedans mellom to busser
   └── .admittansmatrise()        -> Y-buss-matrise (cutsem)

------------------------------------------------------------------------------
grid.Base                   # BaseValues-objektet
grid.Base.Sbase             # S_base [MVA]
grid.Base.Vbase             # V_base

------------------------------------------------------------------------------
 3) BUSSER (liste -> bruk [ ] indeks)
------------------------------------------------------------------------------
grid.bus                    # hele lista med bus-objekter
grid.bus[0]                 # første buss
grid.bus[-1]                # siste buss
grid.bus[2]                 # tredje buss

# Attributter for én buss:
grid.bus[0].name            # busNumber (tallet)
grid.bus[0].Name            # trafostasjon-navn
grid.bus[0].Volt            # spenning [p.u.]
grid.bus[0].Angle           # vinkel [rad]
grid.bus[0].P_gen
grid.bus[0].Q_gen
grid.bus[0].P_load
grid.bus[0].Q_load
grid.bus[0].bidz            # budområde
grid.bus[0].Vbase           # spenningsnivå [kV]
vars(grid.bus[0])           # alt om buss 0 som dict

# Gå gjennom alle busser:
for b in grid.bus:
    print(b.name, b.Volt, b.Angle, b.P_gen, b.P_load)

# Skriv ut alt for alle busser:
for b in grid.bus:
    print(vars(b))

 4) LINJER / GRENER (liste -> bruk [ ] indeks)
------------------------------------------------------------------------------
grid.line                   # hele lista med line-objekter
grid.line[0]                # første linje
grid.line[-1]               # siste linje

# Attributter for én linje:
grid.line[0].lineNumber
grid.line[0].Name           # "bussfra – busstil"
grid.line[0].R
grid.line[0].X
grid.line[0].B
grid.line[0].Frombus
grid.line[0].Tobus
grid.line[0].lenght         # lengde [m]

 5) TRAFOER (liste -> bruk [ ] indeks)
------------------------------------------------------------------------------
grid.trafo                  # hele lista med trafo-objekter
grid.trafo[0]               # første trafo

grid.trafo[0].Name
grid.trafo[0].Type          # "tap"
grid.trafo[0].Ratio
grid.trafo[0].Frombus
grid.trafo[0].Tobus
grid.trafo[0].R
grid.trafo[0].X

------------------------------------------------------------------------------
 6) GENERATORER (liste -> bruk [ ] indeks)
------------------------------------------------------------------------------
grid.gen                    # hele lista med gen-objekter
grid.gen[0]                 # første generator

# Attributter for én generator:
grid.gen[0].name            # generatornavn
grid.gen[0].bus             # buss-id generatoren står på
grid.gen[0].P_max           # Pmax
grid.gen[0].Q_max           # Qmax (test)
grid.gen[0].Q_min           # Qmin (test)
vars(grid.gen[0])           # alt om generator 0 som dict

# Gå gjennom alle generatorer:
for g in grid.gen:
    print(g.name, "bus", g.bus, "Pmax=", g.P_max)


------------------------------------------------------------------------------
 7) NYTTIGE ONELINERS (søk, filtrering, oversikt)
------------------------------------------------------------------------------
# Alle bussnumre:
[b.name for b in grid.bus]

# Alle spenninger:
[b.Volt for b in grid.bus]

# Finn en bestemt buss (f.eks. buss nr 3):
next(b for b in grid.bus if b.name == 3)

# Alle busser med generering (P_gen > 0):
[b.name for b in grid.bus if b.P_gen > 0]

# Alle busser med last (P_load > 0):
[b.name for b in grid.bus if b.P_load > 0]

# Alle linjer koblet til buss 1:
[l.Name for l in grid.line if l.Frombus == 1 or l.Tobus == 1]

# Finn linje mellom buss 1 og 2:
[l for l in grid.line if {l.Frombus, l.Tobus} == {1, 2}]

# Alle generatorer på buss 0:
[g.name for g in grid.gen if g.bus == 0]

# Total last i nettet:
sum(b.P_load for b in grid.bus)

# Total generering i nettet:
sum(b.P_gen for b in grid.bus)
